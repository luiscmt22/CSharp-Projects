C3 Enemy Sprite Library

Pixel sprites for Spectre.Console Canvas. Each grid is a string[]; drop it into a Sprite property in Enemies.json. The dot . = transparent.

How to render (shared)

var palette = new Dictionary<char, Color>
{
    ['D']=new Color(30,62,16),  ['G']=new Color(74,139,44),  ['E']=new Color(255,100,0),
    ['R']=new Color(95,12,12),  ['W']=new Color(238,238,224),['N']=new Color(40,80,22),
    ['B']=new Color(235,232,215),['S']=new Color(175,172,150),['K']=new Color(8,8,10),
    ['C']=new Color(60,120,210), ['A']=new Color(35,80,160),  ['H']=new Color(150,200,245),
};

var sprite = enemy.Sprite;                         // string[] from JSON
var canvas = new Canvas(sprite[0].Length, sprite.Length);
for (int y = 0; y < sprite.Length; y++)
    for (int x = 0; x < sprite[y].Length; x++)
        if (palette.TryGetValue(sprite[y][x], out var color))
            canvas.SetPixel(x, y, color);          // '.' not in palette => transparent
AnsiConsole.Write(canvas);
